
# Ideas for response generation in Flask

```Python
return Response.Ok(data)            # ApiModel objects are serialized as JSON by default
return Response(data)               # 200 implied
return Response.Error(400, error)
return Response.Unauthorized(error) # 401 implied
return Response.Forbidden(error)    # 403 implied
return Response.NotFound(error)     # 404 implied
return Response.Error(error)        # 400 implied

# Fully developed response
return Response(
        StatusCode(200),
        Body(data),
        ContentType.JSON, # == ContentType("application/json")
        Header("X-CSRF-Token", tokeValue),
        Cookie("MyCookieName", "MyCookieValue", httpOnly=True),
)
```
